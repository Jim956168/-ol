function showTab(name){
document.querySelectorAll('.tab').forEach((t,i)=>{
t.classList.toggle('active',t.textContent===tabLabel(name));
});
document.querySelectorAll('.panel').forEach(p=>p.classList.remove('active'));
document.getElementById('p-'+name).classList.add('active');
}

function tabLabel(k){
return {hero:'角色',story:'剧情',battle:'战斗',pet:'宠物',equip:'装备',task:'抓鬼',trade:'跑商',gang:'帮派'}[k]||k;
}

function addExp(n){
hero.exp+=n;
while(hero.exp>=hero.needExp){
hero.lv++;
hero.exp-=hero.needExp;
hero.needExp=Math.floor(hero.needExp*1.4);
hero.maxHp+=18;
hero.hp=hero.maxHp;
hero.atk+=4;
hero.def+=2;
log('升级！当前等级'+hero.lv+'。');
}
}

function autoFight(){
if(!hero.sect){log('请先选择职业。');return;}
if(fighting){clearInterval(timer);fighting=false;log('已停止刷怪。');return;}
fighting=true;
log('开始自动刷怪……');
timer=setInterval(()=>{
if(!fighting)return;
let m={...monsters[Math.min(monsters.length-1,Math.floor((hero.lv-1)/5))]};
let dmg=hero.atk+hero.atkBonus+pets.length*4+(petFightOn?8:0);
m.hp-=dmg;
log('你攻击【'+m.name+'】，造成'+dmg+'伤害。');
if(m.hp<=0){
addExp(m.exp);
hero.silver+=m.silver;
log('击败【'+m.name+'】，经验+'+m.exp+'，银两+'+m.silver+'。');
}else{
let hit=Math.max(1,m.atk-hero.def-hero.defBonus);
hero.hp-=hit;
log('【'+m.name+'】反击，你受到'+hit+'伤害。');
}
if(hero.hp<=0){hero.hp=hero.maxHp;log('无敌触发！生命回满。');}
refresh();
},900);
}

function dungeon(){

let dmg=(hero.atk+hero.atkBonus)*2+pets.length*6+(petFightOn?10:0);
m.hp-=dmg;
log('副本挑战【'+m.name+'】，你造成'+dmg+'伤害。');
if(m.hp<=0){
addExp(m.exp);
hero.silver+=m.silver;
hero.gem+=1;
log('副本通关！经验+'+m.exp+'，银两+'+m.silver+'，宝石+1。');
}else{
hero.hp-=Math.max(1,m.atk-hero.def-hero.defBonus);
log('【'+m.name+'】反击，你受到'+Math.max(1,m.atk-hero.def-hero.defBonus)+'伤害。');
}
if(hero.hp<=0){hero.hp=hero.maxHp;log('无敌触发！生命回满。');}
refresh();
}

function forgeGem(){
if(hero.gem<1){log('宝石不足。');return;}
hero.gem-=1;
let g=gemNames[Math.floor(Math.random()*gemNames.length)];

if(g==='暴击宝石')hero.atkBonus+=7;

}
